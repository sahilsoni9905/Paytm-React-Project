## TECHNICAL FLOW (File by File) 📁

### 1. main.py - The Controller
```python
# User sends: "Hey give my detail"
@app.post("/chat")
async def chat_with_bot(request: ChatRequest):
    # Step 1: Get AI understanding
    intent = await gemini_understand_intent(request.message)
    
    # Step 2: Based on intent, take action
    if intent["action"] == "get_user_details":
        # Step 3: Call your backend
        user_data = await get_user_details_from_backend(request.user_token)
        
        # Step 4: Generate natural response with AI
        response = await gemini_generate_response(user_data, request.message)
        
        return ChatResponse(response=response, action_taken="fetched_user_details")
```

### 2. ai_model.py - The Gemini Brain
```python
async def gemini_understand_intent(user_message):
    """Step 1: Understand what user wants"""
    prompt = f"""
    User said: "{user_message}"
    
    Classify this into:
    - user_details: wants their account info
    - balance: wants balance info  
    - general: just chatting
    
    Respond in JSON format.
    """
    
    # Call Gemini API
    response = client.chat.completions.create(
        model="gemini-2.5-flash",
        messages=[{"role": "user", "content": prompt}]
    )
    
    return json.loads(response.choices[0].message.content)

async def gemini_generate_response(user_data, original_message):
    """Step 4: Create natural response"""
    prompt = f"""
    User asked: "{original_message}"
    User data: {user_data}
    
    Create a friendly, natural response showing their details.
    Be conversational and helpful.
    """
    
    # Call Gemini API for response generation
    response = client.chat.completions.create(
        model="gemini-2.5-flash", 
        messages=[{"role": "user", "content": prompt}]
    )
    
    return response.choices[0].message.content
```

### 3. Backend Call Function
```python
async def get_user_details_from_backend(user_token):
    """Step 3: Get data from your Node.js API"""
    headers = {"Authorization": f"Bearer {user_token}"}
    
    response = requests.get(
        "http://localhost:3000/api/v1/user/get-user-details",
        headers=headers
    )
    
    return response.json()  # Returns: {username, firstName, lastName, profilePic}
```

from fastapi import FastAPI, HTTPException
import requests
import json
from ai_model import gemini_understand_intent, gemini_generate_response

app = FastAPI()

@app.post("/chat")
async def chat_with_bot(request: ChatRequest):
    try:
        # STEP 1: Understand user intent with Gemini
        print(f"🧠 Analyzing: '{request.message}'")
        intent = await gemini_understand_intent(request.message)
        print(f"🎯 Intent: {intent}")
        
        # STEP 2: Take action based on intent
        if intent["action"] == "get_user_details":
            print("📞 Calling backend API...")
            
            # STEP 3: Get data from your backend
            user_data = await get_user_details_from_backend(request.user_token)
            print(f"📋 Got data: {user_data}")
            
            # STEP 4: Generate natural response with Gemini
            print("🤖 Generating response...")
            response = await gemini_generate_response(user_data, request.message, intent)
            print(f"💬 Response: {response}")
            
            return ChatResponse(
                response=response,
                action_taken="fetched_user_details_with_ai"
            )
        
        # Handle other intents...
        
    except Exception as e:
        print(f"❌ Error: {e}")
        raise HTTPException(status_code=500, detail=str(e))
```

from openai import OpenAI

async def gemini_understand_intent(user_message):
    """Use Gemini to understand what user wants"""
    
    system_prompt = """
    You are a Paytm assistant. Analyze user messages and respond in JSON:
    
    {
      "intent": "user_details" | "balance" | "transfer" | "general",
      "action": "get_user_details" | "get_balance" | "transfer_money" | "general_chat", 
      "confidence": "high" | "medium" | "low",
      "tone": "casual" | "formal" | "friendly",
      "specific_request": "what exactly they want"
    }
    
    Examples:
    "hey give my detail" → user_details intent
    "what's my balance" → balance intent  
    "send money to john" → transfer intent
    """
    
    messages = [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": user_message}
    ]
    
    response = client.chat.completions.create(
        model="gemini-2.5-flash",
        response_format={"type": "json_object"},
        messages=messages
    )
    
    return json.loads(response.choices[0].message.content)

async def gemini_generate_response(user_data, original_message, intent):
    """Generate natural response using Gemini"""
    
    prompt = f"""
    Create a natural, helpful response for a Paytm user.
    
    User asked: "{original_message}"
    User's tone: {intent.get('tone', 'friendly')}
    User data: {json.dumps(user_data, indent=2)}
    
    Guidelines:
    - Match the user's tone ({intent.get('tone', 'friendly')})
    - Be helpful and informative
    - Use emojis appropriately  
    - Format the data nicely
    - Ask if they need anything else
    
    Create a response that feels natural and conversational.
    """
    
    response = client.chat.completions.create(
        model="gemini-2.5-flash",
        messages=[{"role": "user", "content": prompt}]
    )
    
    return response.choices[0].message.content.strip()
```

## COMPLETE TECHNICAL FLOW 🔧

### Financial Query Processing Pipeline

#### 1. Date Intelligence 📅
```python
# AI extracts dates from natural language:
"last month" → time_period: "last month"  
"Oct 1 to Oct 15" → date_from: "2025-10-01", date_to: "2025-10-15"
"yesterday" → time_period: "yesterday"
"this week vs last week" → complex comparison query
```

#### 2. Backend API Selection 🎯
```python
# AI chooses correct endpoint:
"how much I spent" → GET /api/v1/user/get-expenses
"money I received" → GET /api/v1/user/get-income  
"my transactions" → GET /api/v1/user/get-transaction-history
"my details" → GET /api/v1/user/get-user-details
```

#### 3. Query Parameter Building 🔨
```python
# From AI extracted info to API params:
extracted_info = {
    "time_period": "this month",
    "transaction_type": "sent"
}

# Becomes:
api_params = {"time_period": "this month"}
endpoint = "/get-expenses"
```

#### 4. Data Processing & Response 📊
```python
# Backend processes dates:
"this month" → startOfMonth = new Date(2025, 9, 1)
           → filter: {createdAt: {$gte: startOfMonth}}

# Returns structured data:
{
  "totalExpenses": 2450.75,
  "transactionCount": 12, 
  "expenseBreakdown": {...},
  "recentTransactions": [...]
}
```

#### 5. Intelligent Response Generation 🤖
```python
# AI creates contextual response:
- Matches user's tone (casual/formal)
- Formats numbers with currency symbols
- Adds insights ("You spent 11% less this week")  
- Suggests follow-up actions
- Uses appropriate emojis
```

## Common Issues & Solutions

### "Module not found" 
→ Run: `pip install -r requirements.txt`

### "Connection refused"
→ Make sure your Node.js backend is running on port 3000

### "Unauthorized" error
→ Make sure you're sending a valid JWT token

### Bot says "I don't understand"
→ Try exact phrases like "What's my name?" or add more keywords
